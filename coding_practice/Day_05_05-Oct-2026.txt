## - Jenkinsfile -- with Git Integration and Testing

```groovy
pipeline {
  agent any
  stages {
    stage('Clone Repository') {
	steps {
	  git 'https://github.com/pathnex/sample-repo.git'
	}
    }
    stage('Build' {
	steps {
	  echo 'Building Project from Git Repository'
	}
    }
    stage('Test') {
	steps {
	  echo 'Running Unit Tests'
	}
    }
    stage('Deploy') {
	steps {
	  echo 'Deploying application'
	}
    }
  }
}

## Ansible -- Create User and Set Permissions

- name: Create a new USer
  hosts: all
  become: yes
  tasks:
    - name: Create USer
      user:
	name: pathnex
	state: present
	shell: /bin/bash
    - name: Set permissions for the user
      file:
	path: /home/pathnex
	owner: pathnex
	group: pathnex
	mode: '0755'
