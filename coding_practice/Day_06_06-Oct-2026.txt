## Ansible -- Install Docker


```yaml
- name: Install Docker
  hosts: all
  become: yes
  tasks:
    - name: Install Docker
      yum:
	name: docker
	state: present
    - name: Start Docker
      service:
	name: docker
	state: started
	enabled: yes

### Kubernets -- Pod With Volume Mount

apiVersion: v1
kind: pod
metadata:
  name: pathnex-pod
spec:
  container:
    - name: web
      image: nginx
      volumeMounts:
	- mountPath: /usr/share/nginx/html
	  name: html-volume
  volumes:
    - name: html-volume
      hostPath:
	path: /data
	type: Directory

### Shell script  -- Check For Running Services

#/bin/bash
services=("nginx" "docker" "httpd")
for service in "${service[@]}"
do
  if systemctl is-actice --quiet $service; then
    echo "$service is running"
  else
    echo "$service is not running"
  fi
done
